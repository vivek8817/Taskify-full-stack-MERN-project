### 🟢 STEP 11: Auth Middleware (Route Protector)

Koi bhi random user tasks access nahi kar sakta. Humein ek middleware chahiye jo check kare ki request ke sath valid JWT token aaya hai ya nahi.

`src/middleware` folder banaiye aur usme `authMiddleware.ts` add karein:

```typescript
// backend/src/middleware/authMiddleware.ts
import jwt from 'jsonwebtoken';
import { Request, Response, NextFunction } from 'express';
import User from '../models/UserModel';

// Express Request type ko extend kar rahe hain taaki req.user inject kar sakein
export interface AuthRequest extends Request {
  user?: any;
}

export const protect = async (req: AuthRequest, res: Response, next: NextFunction): Promise<void> => {
  let token;

  if (req.headers.authorization && req.headers.authorization.startsWith('Bearer')) {
    try {
      // Token extract karo: "Bearer eyJhbG..." -> "eyJhbG..."
      token = req.headers.authorization.split(' ')[1];

      // Token verify karo
      const decoded: any = jwt.verify(token, process.env.JWT_SECRET as string);

      // User ko database se fetch karo (password chhod kar) aur req mein daal do
      req.user = await User.findById(decoded.id).select('-password');

      next();
    } catch (error) {
      res.status(401).json({ message: 'Not authorized, token failed' });
    }
  }

  if (!token) {
    res.status(401).json({ message: 'Not authorized, no token' });
  }
};

```

### 🟢 STEP 13: Task Controller (The CRUD Logic)

Yahan hum Create, Read, Update, aur Delete ke functions likhenge. Dhyaan rakhna, har user ko sirf **apne hi tasks** dikhne chahiye aur wo sirf apne hi tasks delete/update kar sake.

`src/controllers/taskController.ts` banaiye:

```typescript
// backend/src/controllers/taskController.ts
import { Response } from 'express';
import Task from '../models/TaskModel';
import { AuthRequest } from '../middleware/authMiddleware';

// @desc    Get user tasks
// @route   GET /api/tasks
export const getTasks = async (req: AuthRequest, res: Response): Promise<void> => {
  const tasks = await Task.find({ user: req.user.id }).sort({ createdAt: -1 }); // Naye tasks pehle
  res.status(200).json(tasks);
};

// @desc    Create a task
// @route   POST /api/tasks
export const createTask = async (req: AuthRequest, res: Response): Promise<void> => {
  if (!req.body.title) {
    res.status(400).json({ message: 'Please add a task title' });
    return;
  }

  const task = await Task.create({
    title: req.body.title,
    user: req.user.id,
  });

  res.status(201).json(task);
};

// @desc    Update a task (toggle completion/edit text)
// @route   PUT /api/tasks/:id
export const updateTask = async (req: AuthRequest, res: Response): Promise<void> => {
  const task = await Task.findById(req.params.id);

  if (!task) {
    res.status(404).json({ message: 'Task not found' });
    return;
  }

  // Check karo ki yeh task usi user ka hai ya nahi
  if (task.user.toString() !== req.user.id) {
    res.status(401).json({ message: 'User not authorized' });
    return;
  }

  const updatedTask = await Task.findByIdAndUpdate(req.params.id, req.body, {
    new: true, // Updated document return karega
  });

  res.status(200).json(updatedTask);
};

// @desc    Delete a task
// @route   DELETE /api/tasks/:id
export const deleteTask = async (req: AuthRequest, res: Response): Promise<void> => {
  const task = await Task.findById(req.params.id);

  if (!task) {
    res.status(404).json({ message: 'Task not found' });
    return;
  }

  // Ownership check
  if (task.user.toString() !== req.user.id) {
    res.status(401).json({ message: 'User not authorized' });
    return;
  }

  await task.deleteOne();
  
  res.status(200).json({ id: req.params.id });
};

```

### 🟢 STEP 15: Task API Layer (`src/api/tasks.ts`)

Frontend se backend ko request bhejte waqt humein ab apna JWT token bhi sath bhejna hoga.

`src/api` folder mein ek nayi file banaiye `tasks.ts`:

```typescript
// src/api/tasks.ts
import axios from 'axios';

const API_URL = 'http://localhost:5000/api/tasks';

export interface Task {
  _id: string;
  title: string;
  completed: boolean;
  createdAt: string;
}

// Token attach karne ke liye helper function
const getAuthHeaders = () => {
  const user = JSON.parse(localStorage.getItem('taskify_user') || '{}');
  return {
    headers: {
      Authorization: `Bearer ${user.token}`,
    },
  };
};

export const taskAPI = {
  getTasks: async (): Promise<Task[]> => {
    const response = await axios.get(API_URL, getAuthHeaders());
    return response.data;
  },

  createTask: async (title: string): Promise<Task> => {
    const response = await axios.post(API_URL, { title }, getAuthHeaders());
    return response.data;
  },

  updateTask: async (id: string, taskData: Partial<Task>): Promise<Task> => {
    const response = await axios.put(`${API_URL}/${id}`, taskData, getAuthHeaders());
    return response.data;
  },

  deleteTask: async (id: string): Promise<{ id: string }> => {
    const response = await axios.delete(`${API_URL}/${id}`, getAuthHeaders());
    return response.data;
  }
};

```

        setLoading(false);
      }

      const newTask = await taskAPI.createTask(title);
      setTasks([newTask, ...tasks]); // Optimistically prepend new task
    } catch (error) {
      console.error("Failed to add task", error);
